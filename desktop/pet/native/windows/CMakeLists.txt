cmake_minimum_required(VERSION 3.24)
project(tulpa_windows_native LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /EHsc /utf-8 /permissive-)
endif()
enable_testing()
add_executable(audio-buffer-test audio_buffer.cpp audio_buffer_test.cpp)
add_test(NAME audio-buffer COMMAND audio-buffer-test)
add_executable(protocol-state-test tests/protocol_state_test.cpp)
add_test(NAME protocol-state COMMAND protocol-state-test)
if(MSVC)
  target_compile_options(audio-buffer-test PRIVATE /UNDEBUG)
  target_compile_options(protocol-state-test PRIVATE /UNDEBUG)
else()
  target_compile_options(audio-buffer-test PRIVATE -UNDEBUG)
  target_compile_options(protocol-state-test PRIVATE -UNDEBUG)
endif()
if(NOT WIN32)
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows native helpers currently require x64")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
find_package(nlohmann_json 3.11.3 EXACT CONFIG QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(json)
endif()
add_executable(fish-memory-perception-helper helper_main.cpp screen_capture.cpp computer_history.cpp audio.cpp audio_buffer.cpp)
target_link_libraries(fish-memory-perception-helper PRIVATE nlohmann_json::nlohmann_json windowsapp d3d11 dxgi windowscodecs ole32 oleaut32 user32 advapi32 crypt32 uiautomationcore uuid mmdevapi)
if(NOT DEFINED NODE_INCLUDE_DIR OR NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "Set NODE_INCLUDE_DIR to the Electron/Node headers containing node_api.h")
endif()
if(NOT DEFINED NODE_LIBRARY OR NOT EXISTS "${NODE_LIBRARY}")
  message(FATAL_ERROR "Set NODE_LIBRARY to the matching Electron x64 node.lib")
endif()
add_library(pet-application-activation SHARED input_bridge.cpp delay_load_hook.cpp)
set_target_properties(pet-application-activation PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(pet-application-activation PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_definitions(pet-application-activation PRIVATE NAPI_VERSION=8)
target_link_libraries(pet-application-activation PRIVATE "${NODE_LIBRARY}" user32 ole32 uuid delayimp)
target_link_options(pet-application-activation PRIVATE /DELAYLOAD:node.exe)
install(TARGETS fish-memory-perception-helper pet-application-activation RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES nlohmann-json-LICENSE.txt DESTINATION licenses)
install(FILES audio-LICENSE.txt DESTINATION licenses)
install(FILES node-gyp-LICENSE.txt NOTICES.md DESTINATION licenses)
